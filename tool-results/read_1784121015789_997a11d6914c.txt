   929→        // Typewriter Effect
   930→        function initTypewriter() {
   931→            const words = [
   932→                'aplicações web modernas',
   933→                'soluções corporativas',
   934→                'dashboards interativos',
   935→                'integrações SAP/TMS',
   936→                'APIs escaláveis',
   937→                'experiências incríveis'
   938→            ];
   939→            
   940→            let wordIndex = 0;
   941→            let charIndex = 0;
   942→            let isDeleting = false;
   943→            const typewriterElement = document.getElementById('typewriter');
   944→            
   945→            if (!typewriterElement) return;
   946→            
   947→            function type() {
   948→                const currentWord = words[wordIndex];
   949→                
   950→                if (isDeleting) {
   951→                    typewriterElement.textContent = currentWord.substring(0, charIndex - 1);
   952→                    charIndex--;
   953→                } else {
   954→                    typewriterElement.textContent = currentWord.substring(0, charIndex + 1);
   955→                    charIndex++;
   956→                }
   957→                
   958→                if (!isDeleting && charIndex === currentWord.length) {
   959→                    isDeleting = true;
   960→                    setTimeout(type, 2000);
   961→                    return;
   962→                }
   963→                
   964→                if (isDeleting && charIndex === 0) {
   965→                    isDeleting = false;
   966→                    wordIndex = (wordIndex + 1) % words.length;
   967→                    setTimeout(type, 500);
   968→                    return;
   969→                }
   970→                
   971→                const speed = isDeleting ? 50 : 100;
   972→                setTimeout(type, speed);
   973→            }
   974→            
   975→            type();
   976→        }
